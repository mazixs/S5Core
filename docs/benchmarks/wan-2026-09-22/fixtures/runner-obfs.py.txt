def stop(host,name):

 ssh(host,f'''p=$(cat {r}/{name}.pid 2>/dev/null) || exit 0
case "$(readlink /proc/$p/exe 2>/dev/null)" in {r}/*) kill -TERM "$p" 2>/dev/null || true;; *) rm -f {r}/{name}.pid; exit 0;; esac

def snapshot(host):
 r=REMOTE[host]
 return ssh(host,f'''head -1 /proc/stat
