#!/usr/bin/env bash
# Runs inside netem.sh: captures the tunnel leg while one matrix run goes. usage: cap.sh MODE OUTDIR PCAP ONLY
set -u
cd "$(git rev-parse --show-toplevel)/bench/matrix"
tcpdump -Z root -i lo -n -s 200 -w "$3" 'tcp port 41443 or tcp port 41444' 2>"$3.err" &
tp=$!
sleep 1
./run.sh "$1" "$2" -scale 0.3 -only "$4"
rc=$?
sleep 1
kill "$tp"
wait "$tp" 2>/dev/null
exit $rc
