package socks5

import (
	"context"
	"errors"
	"fmt"
	"net"
	"testing"
	"testing/synctest"
	"time"
)

func TestReviewFallbackPrematurelyAbortsWorkingIP(t *testing.T) {
	synctest.Test(t, func(t *testing.T) {
		dial := func(ctx context.Context, _, addr string) (net.Conn, error) {
			if addr != "192.0.2.1:443" {
				return nil, errors.New("connection refused")
			}
			select {
			case <-time.After(time.Second):
				a, b := net.Pipe()
				b.Close()
				return a, nil
			case <-ctx.Done():
				return nil, ctx.Err()
			}
		}
		ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
		defer cancel()
		start := time.Now()
		direct, e := dial(ctx, "tcp", "192.0.2.1:443")
		if e != nil {
			t.Fatal(e)
		}
		direct.Close()
		t.Logf("old single-address behavior: success in %s", time.Since(start))
		ctx2, cancel2 := context.WithTimeout(context.Background(), 10*time.Second)
		defer cancel2()
		candidates := make([]dialCandidate, 16)
		for i := range candidates {
			candidates[i] = dialCandidate{ctx: ctx2, addr: fmt.Sprintf("192.0.2.%d:443", i+1)}
		}
		start = time.Now()
		c, e := dialResolved(ctx2, dial, candidates)
		if c != nil {
			c.Close()
		}
		t.Logf("new fallback: elapsed=%s err=%v parent_err=%v", time.Since(start), e, ctx2.Err())
		if e == nil {
			t.Fatal("expected current premature timeout")
		}
		if time.Since(start) != 625*time.Millisecond || ctx2.Err() != nil {
			t.Fatal("unexpected timing")
		}
	})
}
