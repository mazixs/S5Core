Run from the S5Core checkout. This creates an isolated temporary release tree.
The six diagnostic tests assert the observed defect, so they pass before a fix.
No remote access, configuration change or service restart is performed here.

audit_repo="$PWD"
audit_data="$audit_repo/docs/reports/data-v2.3-audit-2026-10-03"
audit_copy=$(mktemp -d /tmp/s5core-release-audit-repeat.XXXXXX)
git archive v2.3.0 | tar -x -C "$audit_copy"
cd "$audit_copy"
./scripts/pre-commit.sh

cp "$audit_data/socks5-probes.go.txt" internal/socks5/audit_release_test.go
