#!/usr/bin/env bash
# 2.1.0 vs the tree where suite8 could not tell: the burst on a fresh UDP association
# (udp/stream-40mbit, udp/sweep-40mbit; RTT 40 ms, no loss, 10 rounds - its loss is bimodal per run)
# and HTTP/3 bulk under 0.5% loss (h3/upload, h3/large; 6 rounds).
set -uo pipefail
cd "$(dirname "$0")"
out=${OUT:-results10}
mkdir -p "$out"
export AUTH=${AUTH:-member} FLUSH=60s WARMUP=1
ONLY="udp/stream-40mbit,udp/sweep-40mbit"
declare -A D=([obfs]="0,8" [wss]="5,13") A=([obfs]="1-2,9-10" [wss]="6,14") B=([obfs]="3-4,11-12" [wss]="7,15")

pair() { # round transport
	local r=$1 m=$2 ca=${A[$2]} cb=${B[$2]} loss=${LOSS:-0}
	((r % 2 == 0)) && { ca=${B[$m]}; cb=${A[$m]}; }
	taskset -c "${D[$m]}" ./netem.sh 40 "$loss" 200 all ./run.sh direct "$out/direct-$m-$r" -scale 0.3 -only "$ONLY" > "$out/direct-$m-$r.log" 2>&1 &
	TAG=v210 taskset -c "$ca" ./netem.sh 40 "$loss" 200 tunnel ./run.sh "$m" "$out/v210-$m-$r" -scale 0.3 -only "$ONLY" > "$out/v210-$m-$r.log" 2>&1 &
	TAG=cur taskset -c "$cb" ./netem.sh 40 "$loss" 200 tunnel ./run.sh "$m" "$out/cur-$m-$r" -scale 0.3 -only "$ONLY" > "$out/cur-$m-$r.log" 2>&1 &
	wait
}

rounds() { # count
	for r in $(seq 1 "$1"); do
		pair "$r" obfs & pair "$r" wss & wait
		echo "$out round $r done: $(for d in "$out"/*-"$r"; do printf '%s=%s ' "${d##*/}" "$(grep -o '"rc":[0-9]*' "$d/run.json" 2>/dev/null || echo none)"; done)"
	done
}
rounds 10
out=${out}-h3 ONLY="h3/upload,h3/large" LOSS=0.5
mkdir -p "$out"
rounds 6
echo suite10-done
