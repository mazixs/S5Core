#!/usr/bin/env python3
"""Per-run numbers from suite10.sh: compare10.py RESULTS (results10 or results10-h3)"""
import json, sys
from pathlib import Path
from statistics import median

root = Path(sys.argv[1])
names = sorted({n for f in root.glob('*/result.json') for n in json.loads(f.read_text())['result']})
bulk = all(n.startswith('h3/') for n in names)
if bulk:
    print('| Сценарий | Путь | Вариант | Прогонов | МБ/с, медиана | МБ/с по прогонам |')
    print('|---|---|---|---:|---:|---|')
else:
    print('| Сценарий | Путь | Вариант | Прогонов | Потери, медиана | Прогонов с потерями >5% | Потери по прогонам, % | p99 мс, медиана |')
    print('|---|---|---|---:|---:|---:|---|---:|')
for name in names:
    for m in ('obfs', 'wss'):
        for v in ('direct', 'v210', 'cur'):
            loss, p99 = [], []
            for d in sorted(root.glob(f'{v}-{m}-*'), key=lambda p: int(p.name.rsplit('-', 1)[1]) if p.is_dir() else 0):
                f = d / 'result.json'
                if not d.is_dir() or not f.exists():
                    continue
                s = json.loads(f.read_text())['result'].get(name)
                if s:
                    loss.append(s.get('mbps', 0) if bulk else (s.get('extra') or {}).get('loss_pct', 0))
                    p99.append(s['p99_ms'])
            if loss and bulk:
                print(f'| `{name}` | {m} | {v} | {len(loss)} | {median(loss):.1f} | ' + ', '.join(f'{l:.1f}' for l in loss) + ' |')
            elif loss:
                print(f'| `{name}` | {m} | {v} | {len(loss)} | {median(loss):.2f}% | {sum(l > 5 for l in loss)} | '
                      + ', '.join(f'{l:.1f}' for l in loss) + f' | {median(p99):.0f} |')
