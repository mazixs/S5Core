package main

import ("strings"; "golang.org/x/net/idna")

type domainMatcher struct { patterns []string; exact map[string]struct{} }
func newDomainMatcher(patterns []string) *domainMatcher { m := &domainMatcher{patterns: patterns}; if len(patterns)>0 { m.exact=map[string]struct{}{"": {}} }; return m }
func (m *domainMatcher) Match(fqdn string) bool { if m == nil { return false }; return matchDomain(fqdn,m.patterns) }

func matchDomain(fqdn string, patterns []string) bool {
	fqdn = strings.ToLower(fqdn)
	if ascii, err := idna.ToASCII(fqdn); err == nil {
		fqdn = ascii
	}
	for _, p := range patterns {
		p = strings.ToLower(strings.TrimSpace(p))
		if ascii, err := idna.ToASCII(p); err == nil {
			p = ascii
		}

		if p == fqdn {
			return true
		}

		// Wildcard match: *.example.com matches sub.example.com and deep.sub.example.com
		if strings.HasPrefix(p, "*.") {
			suffix := p[1:] // ".example.com"
			if strings.HasSuffix(fqdn, suffix) {
				return true
			}
			// Also match the base domain itself
			if fqdn == p[2:] {
				return true
			}
		}
	}
	return false
}
